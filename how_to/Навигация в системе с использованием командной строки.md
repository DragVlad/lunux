```bash
# Вывод пользователя
whoami

# Текущая директория
pwd

# Вывод содержимого директории
ls
ls -la # подробный вывод, в том числе и скрытых файлов

# Вывод содержимого файла
cat

# Смена директории
cd

# Очистка экрана терминала
clear # или CTRL+l

head -n 10 file.txt # вывод первых 10 строчек файла
tail -n 10 file.txt # вывод первых 10 строчек файла

less filename.txt # просмотр содержимого файла + при нажатии / и нужного слова показывает его
```
