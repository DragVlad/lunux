# База консоли

## Файлы/перемещение/создание и тп.
```bash
ls -lah # показать содержимое директории
cd directoryname # переход в директорию
pwd # в какомй директории мы находимся

mkdir directoryname # создание директории
mkdir dir1 dir2 dir3 # создание нескольких директорий
mkdir -p dir1/dir2 # создание вложенных директорий

rm -r dir1 # удаление директории
rm file* # удаление всех совпадений
rm -rf dir1 # жесткое удаление директории

cp file1 file2 # копирование файла
cp -r dir1 dir2 # копирование директории

mv dir1/file1 dir2/file2 # перемещение файла откуда/куда

touch filename # создание файла

cat filename # вывод содержимого файла

less filename # постраничное чтение файла

head -n 5 filename # вывод первых 5 строк файла
tail -n 5 filename # вывод первых 5 строк файла
tail -fn 5 filename # вывод первых 5 и динамическое отображение изменений

echo "" > testfile # запись строки в файл
```

## Поиск чего-либо

```bash
grep ssh /var/log/auth.log | grep username # вывести список строк с вхождением ssh и пользователем username
grep -i " username" dirname # не регистрозависимый поиск username
grep -v username dirname # все строки где нет username
grep -Ri user dirname # рекурсивный поиск строки по файлам в директории

find dirname -name filename* # поиск директории файла по совпдаению названия файла
find dirname -name *.log # поиск всех лог файлов в dirname
```

## База по процессам
```bash
ps aux # вывод всех процессов
ps aux | grep nginx # вывод всех процессов по nginx
ps -C nginx

kill -9 333444 # жесткое выключение процесса
```

## База по сети
```bash
apt-get install net-tools # Установка базового пакета тулзов
# пинг тупой эхо-реквест БЕЗ HTTP
ping ya.ru -c 4 # 4 пакета яндексу - проверка доступности/маршрутизации
ifconfig # штука из пакета networktools для показа конфигурации сетевых интерфейсов
ip a # тоже самое что и ifconfig
sudo netstat -tuln # t-TCP u-UDP l-lesson n-что-то полезное, показывает все открытые порты
sudo netstat -tulnp # все открытые порты + их пиды-демоны
ss -antlp
```

## Анализ дисков
```bash
df -h # просмотр занятости дисков
lsblk # блочные устройства(диски)
du -h -d 1 /var # занятость директории var
sudo lsof -p PID_NUM # показывает все файлы с которыми работает данный процесс

# du и df по разному оценивают занятое место, du может не показывать по факту что и сколько занимает
# пример такого кейса
ping ya.ru > filename $ # фоновая задача
# [1]3322
sudo lsof -p 3322 # смотрим файлы ping
rm filename
# в диск все-равно пишетутся данные
sudo lsof | grep deleted # находим сломанную запись в файл
kill -9 3322
```
