# Пакеты

```bash
cat /etc/*release # покажет систему

apt search vim # поиск по вхождению
apt -a show vim # покажет все версии пакета
sudo apt install tree # установка пакета
sudo dpkg -l | grep libc6 # поиск пакета в dpkg
sudo apt remove tree # удаление пакета нетрогая конфигурационные файлы
sudo apt purge tree # удаляет все что связанно с этим пакетом
sudo apt autoremove # удаление ненужных пакетов
```

```bash
curl -L -O http://ftp.de.debian.org/debian/pool/main/h/htop/htop_3.0.5-7_amd64.deb # скачивание пакета
sudo dpkg -i htop_3.0.5-7_amd64.deb # установка пакета
sudo dpkg -l | grep htop # поиск пакета
sudo dpkg -r htop # удаление пакета
sudo dpkg --purge htop # жесткое удаление пакета
```
```bash
apt download vim # скачивание пакета
dpkg-deb -x vim_2%3a9.1.0016-1ubuntu7.11_amd64.deb ./vim # достаем все содержимое из deb пакета в директорию
dpkg-deb -e vim_2%3a9.1.0016-1ubuntu7.11_amd64.deb ./vim-meta # достаем мета файлы
```
